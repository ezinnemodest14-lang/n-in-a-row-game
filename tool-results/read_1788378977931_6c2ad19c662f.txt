    39→export function checkWinAt(board, n, winLen, r, c, player) {
    40→  for (const [dr, dc] of DIRS) {
    41→    let count = 1;
    42→    let rr = r + dr,
    43→      cc = c + dc;
    44→    while (inBounds(n, rr, cc) && board[idx(n, rr, cc)] === player) {
    45→      count++;
    46→      rr += dr;
    47→      cc += dc;
    48→    }
    49→    rr = r - dr;
    50→    cc = c - dc;
    51→    while (inBounds(n, rr, cc) && board[idx(n, rr, cc)] === player) {
    52→      count++;
    53→      rr -= dr;
    54→      cc -= dc;
    55→    }
    56→    if (count >= winLen) return true;
    57→  }
    58→  return false;
    59→}

    66→export const SCORE_WEIGHTS = {
    67→  triple: 1,
    68→  four: 5,
    69→  five: 15,
    70→  fullRow: 10,
    71→  fullCol: 10,
    72→  fullDiag: 20,
    73→  cross: 3,
    74→  linebreak: 2,
    75→};
    76→const SCORE_KEYS = Object.keys(SCORE_WEIGHTS);

    78→export function scoreMoveFeatures(board, n, row, col, player) {
    79→  const opp = player === 1 ? 2 : 1;
    80→  const feat = Object.fromEntries(SCORE_KEYS.map((k) => [k, 0]));
    81→  let horiz3 = false,
    82→    vert3 = false;
    83→  for (const [dr, dc] of DIRS) {
    84→    let left = 0,
    85→      r = row - dr,
    86→      c = col - dc;
    87→    while (inBounds(n, r, c) && board[idx(n, r, c)] === player) {
    88→      left++;
    89→      r -= dr;
    90→      c -= dc;
    91→    }
    92→    let right = 0;
    93→    r = row + dr;
    94→    c = col + dc;
    95→    while (inBounds(n, r, c) && board[idx(n, r, c)] === player) {
    96→      right++;
    97→      r += dr;
    98→      c += dc;
    99→    }
   100→    const runLen = left + 1 + right,
   101→      offset = left;
   102→    for (const [K, key] of [
   103→      [3, "triple"],
   104→      [4, "four"],
   105→      [5, "five"],
   106→    ]) {
   107→      if (runLen >= K) {
   108→        const startMin = Math.max(0, offset - (K - 1));
   109→        const startMax = Math.min(offset, runLen - K);
   110→        feat[key] += Math.max(0, startMax - startMin + 1);
   111→      }
   112→    }
   113→    if (dr === 0 && dc === 1 && runLen >= 3) horiz3 = true;
   114→    if (dr === 1 && dc === 0 && runLen >= 3) vert3 = true;
   115→    const l1r = row - dr,
   116→      l1c = col - dc,
   117→      r1r = row + dr,
   118→      r1c = col + dc;
   119→    if (
   120→      inBounds(n, l1r, l1c) &&
   121→      inBounds(n, r1r, r1c) &&
   122→      board[idx(n, l1r, l1c)] === opp &&
   123→      board[idx(n, r1r, r1c)] === opp
   124→    ) {
   125→      feat.linebreak += 1;
   126→    }
   127→  }
   128→  if (horiz3 && vert3) feat.cross = 1;
   129→  const cellVal = (r, c) => (r === row && c === col ? player : board[idx(n, r, c)]);
   130→  let fullRow = true;
   131→  for (let c = 0; c < n; c++)
   132→    if (cellVal(row, c) !== player) {
   133→      fullRow = false;
   134→      break;
   135→    }
   136→  let fullCol = true;
   137→  for (let r = 0; r < n; r++)
   138→    if (cellVal(r, col) !== player) {
   139→      fullCol = false;
   140→      break;
   141→    }
   142→  feat.fullRow = fullRow ? 1 : 0;
   143→  feat.fullCol = fullCol ? 1 : 0;
   144→  let fullDiag = 0;
   145→  if (row === col) {
   146→    let ok = true;
   147→    for (let i = 0; i < n; i++)
   148→      if (cellVal(i, i) !== player) {
   149→        ok = false;
   150→        break;
   151→      }
   152→    if (ok) fullDiag += 1;
   153→  }
   154→  if (row + col === n - 1) {
   155→    let ok = true;
   156→    for (let i = 0; i < n; i++)
   157→      if (cellVal(i, n - 1 - i) !== player) {
   158→        ok = false;
   159→        break;
   160→      }
   161→    if (ok) fullDiag += 1;
   162→  }
   163→  feat.fullDiag = fullDiag;
   164→  return feat;
   165→}

   167→export function scoreMoveDelta(board, n, row, col, player) {
   168→  const f = scoreMoveFeatures(board, n, row, col, player);
   169→  return SCORE_KEYS.reduce((s, k) => s + f[k] * SCORE_WEIGHTS[k], 0);
   170→}

   172→export function computeFullScore(board, n) {
   173→  const totals = { 1: 0, 2: 0 };
   174→  for (let p = 1; p <= 2; p++) {
   175→    const seen = { triple: new Set(), four: new Set(), five: new Set() };
   176→    for (let r = 0; r < n; r++) {
   177→      for (let c = 0; c < n; c++) {
   178→        if (board[idx(n, r, c)] !== p) continue;
   179→        for (const [dr, dc] of DIRS) {
   180→          for (const [K, key] of [
   181→            [3, "triple"],
   182→            [4, "four"],
   183→            [5, "five"],
   184→          ]) {
   185→            let ok = true;
   186→            const cells = [];
   187→            for (let k = 0; k < K; k++) {
   188→              const rr = r + dr * k,
   189→                cc = c + dc * k;
   190→              if (!inBounds(n, rr, cc) || board[idx(n, rr, cc)] !== p) {
   191→                ok = false;
   192→                break;
   193→              }
   194→              cells.push(idx(n, rr, cc));
   195→            }
   196→            if (ok) seen[key].add(cells.join(","));
   197→          }
   198→        }
   199→      }
   200→    }
   201→    let total =
   202→      seen.triple.size * SCORE_WEIGHTS.triple + seen.four.size * SCORE_WEIGHTS.four + seen.five.size * SCORE_WEIGHTS.five;
   203→    for (let r = 0; r < n; r++) {
   204→      for (let c = 0; c < n; c++) {
   205→        if (board[idx(n, r, c)] !== p) continue;
   206→        let h = false,
   207→          v = false;
   208→        for (const [dr, dc, flag] of [
   209→          [0, 1, "h"],
   210→          [1, 0, "v"],
   211→        ]) {
   212→          let left = 0,
   213→            rr = r - dr,
   214→            cc = c - dc;
   215→          while (inBounds(n, rr, cc) && board[idx(n, rr, cc)] === p) {
   216→            left++;
   217→            rr -= dr;
   218→            cc -= dc;
   219→          }
   220→          let right = 0;
   221→          rr = r + dr;
   222→          cc = c + dc;
   223→          while (inBounds(n, rr, cc) && board[idx(n, rr, cc)] === p) {
   224→            right++;
   225→            rr += dr;
   226→            cc += dc;
   227→          }
   228→          if (left + 1 + right >= 3) {
   229→            if (flag === "h") h = true;
   230→            else v = true;
   231→          }
   232→        }
   233→        if (h && v) total += SCORE_WEIGHTS.cross;
   234→      }
   235→    }
   236→    for (let r = 0; r < n; r++) {
   237→      let full = true;
   238→      for (let c = 0; c < n; c++)
   239→        if (board[idx(n, r, c)] !== p) {
   240→          full = false;
   241→          break;
   242→        }
   243→      if (full) total += SCORE_WEIGHTS.fullRow;
   244→    }
   245→    for (let c = 0; c < n; c++) {
   246→      let full = true;
   247→      for (let r = 0; r < n; r++)
   248→        if (board[idx(n, r, c)] !== p) {
   249→          full = false;
   250→          break;
   251→        }
   252→      if (full) total += SCORE_WEIGHTS.fullCol;
   253→    }
   254→    let mainOk = true;
   255→    for (let i = 0; i < n; i++)
   256→      if (board[idx(n, i, i)] !== p) {
   257→        mainOk = false;
   258→        break;
   259→      }
   260→    if (mainOk) total += SCORE_WEIGHTS.fullDiag;
   261→    let antiOk = true;
   262→    for (let i = 0; i < n; i++)
   263→      if (board[idx(n, i, n - 1 - i)] !== p) {
   264→        antiOk = false;
   265→        break;
   266→      }
   267→    if (antiOk) total += SCORE_WEIGHTS.fullDiag;
   268→    totals[p] = total;
   269→  }
   270→  return totals;
   271→}

   277→export const CAT = {
   278→  NONE: 0,
   279→  ONE: 1,
   280→  TWO: 2,
   281→  SIMPLE_THREE: 3,
   282→  BROKEN_THREE: 4,
   283→  OPEN_THREE: 5,
   284→  SIMPLE_FOUR: 6,
   285→  OPEN_FOUR: 7,
   286→  WIN: 8,
   287→};
   288→export const CAT_NAME = {
   289→  0: "None",
   290→  1: "One",
   291→  2: "Two",
   292→  3: "Simple three",
   293→  4: "Broken three",
   294→  5: "Open three",
   295→  6: "Simple four",
   296→  7: "Open four",
   297→  8: "Five — wins",
   298→};
   299→export const CAT_VALUE = { 0: 0, 1: 5, 2: 50, 3: 300, 4: 1200, 5: 3000, 6: 8000, 7: 20000, 8: 1000000 };

   301→function classifyDirection(board, n, row, col, dr, dc, player, winLen) {
   302→  let left = 0,
   303→    r = row - dr,
   304→    c = col - dc;
   305→  while (inBounds(n, r, c) && board[idx(n, r, c)] === player) {
   306→    left++;
   307→    r -= dr;
   308→    c -= dc;
   309→  }
   310→  const leftOpen = inBounds(n, r, c) && board[idx(n, r, c)] === 0;
   311→  let right = 0;
   312→  r = row + dr;
   313→  c = col + dc;
   314→  while (inBounds(n, r, c) && board[idx(n, r, c)] === player) {
   315→    right++;
   316→    r += dr;
   317→    c += dc;
   318→  }
   319→  const rightOpen = inBounds(n, r, c) && board[idx(n, r, c)] === 0;
   320→  const runLen = left + 1 + right;
   321→  const gap = winLen - runLen;
   322→
   323→  if (runLen >= winLen) return CAT.WIN;
   324→  if (gap === 1) {
   325→    if (leftOpen && rightOpen) return CAT.OPEN_FOUR;
   326→    if (leftOpen || rightOpen) return CAT.SIMPLE_FOUR;
   327→    return CAT.NONE;
   328→  }
   329→  if (gap === 2) {
   330→    if (!leftOpen && !rightOpen) return CAT.NONE;
   331→    if (leftOpen && rightOpen) {
   332→      const leftExtendOpen = (() => {
   333→        const rr = row - dr * (left + 2),
   334→          cc = col - dc * (left + 2);
   335→        return inBounds(n, rr, cc) && board[idx(n, rr, cc)] === 0;
   336→      })();
   337→      const rightExtendOpen = (() => {
   338→        const rr = row + dr * (right + 2),
   339→          cc = col + dc * (right + 2);
   340→        return inBounds(n, rr, cc) && board[idx(n, rr, cc)] === 0;
   341→      })();
   342→      return leftExtendOpen || rightExtendOpen ? CAT.OPEN_THREE : CAT.BROKEN_THREE;
   343→    }
   344→    return CAT.SIMPLE_THREE;
   345→  }
   346→  if (gap === 3) return runLen >= 2 ? CAT.TWO : CAT.ONE;
   347→  if (runLen >= 1) return CAT.ONE;
   348→  return CAT.NONE;
   349→}

   351→export function classifyCell(board, n, row, col, player, winLen) {
   352→  let best = CAT.NONE,
   353→    second = CAT.NONE,
   354→    strongDirs = 0,
   355→    bestDir = null;
   356→  for (const [dr, dc] of DIRS) {
   357→    const cat = classifyDirection(board, n, row, col, dr, dc, player, winLen);
   358→    if (cat >= CAT.OPEN_THREE) strongDirs++;
   359→    if (cat > best) {
   360→      second = best;
   361→      best = cat;
   362→      bestDir = [dr, dc];
   363→    } else if (cat > second) {
   364→      second = cat;
   365→    }
   366→  }
   367→  return {
   368→    category: best,
   369→    secondCategory: second,
   370→    isFork: strongDirs >= 2,
   371→    score: CAT_VALUE[best] + 0.3 * CAT_VALUE[second],
   372→    direction: bestDir,
   373→  };
   374→}

   376→export function classifyBoard(board, n, player, winLen) {
   377→  const out = new Map();
   378→  for (let i = 0; i < board.length; i++) {
   379→    if (board[i] !== 0) continue;
   380→    const row = Math.floor(i / n),
   381→      col = i % n;
   382→    out.set(i, classifyCell(board, n, row, col, player, winLen));
   383→  }
   384→  return out;
   385→}

   387→// Signed static eval: POSITIVE always means the human player is ahead.
   388→// This exact sign was inverted for a long time in the reference project
   389→// (rebuild-spec bug #15) — kept as one single, explicit, tested formula.
   390→export function evaluatePosition(board, n, winLen) {
   391→  function sideTotal(player) {
   392→    let total = 0;
   393→    for (let i = 0; i < board.length; i++) {
   394→      if (board[i] !== 0) continue;
   395→      const row = Math.floor(i / n),
   396→        col = i % n;
   397→      const c = classifyCell(board, n, row, col, player, winLen);
   398→      total += CAT_VALUE[c.category] * 0.001 + CAT_VALUE[c.secondCategory] * 0.0003;
   399→    }
   400→    return total;
   401→  }
   402→  const playerTotal = sideTotal(1);
   403→  const aiTotal = sideTotal(2);
   404→  const diff = playerTotal - aiTotal;
   405→  return { raw: diff, display: Math.max(-100, Math.min(100, Math.round(Math.tanh(diff / 40) * 100))) };
   406→}

   411→export function tacticalMove(board, n, winLen, me, opp) {
   412→  const myCells = classifyBoard(board, n, me, winLen);
   413→  const oppCells = classifyBoard(board, n, opp, winLen);
   414→
   415→  for (const [cell, c] of myCells) if (c.category === CAT.WIN) return { cell, reason: "win" };
   416→  for (const [cell, c] of oppCells) if (c.category === CAT.WIN) return { cell, reason: "block-win" };
   417→  for (const [cell, c] of myCells) if (c.category === CAT.OPEN_FOUR) return { cell, reason: "open-four" };
   418→  for (const [cell, c] of oppCells) if (c.category === CAT.SIMPLE_FOUR) return { cell, reason: "block-four" };
   419→  for (const [cell, c] of oppCells) if (c.category === CAT.OPEN_FOUR) return { cell, reason: "block-open-four" };
   420→  let bestFork = null;
   421→  for (const [cell, c] of myCells) {
   422→    if (c.isFork && (!bestFork || c.score > bestFork.score)) bestFork = { cell, score: c.score };
   423→  }
   424→  if (bestFork) return { cell: bestFork.cell, reason: "fork" };
   425→  for (const [cell, c] of oppCells) if (c.category === CAT.OPEN_THREE) return { cell, reason: "block-open-three" };
   426→  return null;
   427→}

   429→// ---------------------------------------------------------------------------
   430→// GlobalRAVE — per-(boardSize,mode) cross-move / cross-game memory.
   431→// ---------------------------------------------------------------------------
   432→export class GlobalRAVE {
   433→  constructor(n) {
   434→    this.n = n;
   435→    this.visits = { 1: new Float64Array(n * n), 2: new Float64Array(n * n) };
   436→    this.wins = { 1: new Float64Array(n * n), 2: new Float64Array(n * n) };
   437→  }
   438→  update(player, cell, reward) {
   439→    this.visits[player][cell] += 1;
   440→    this.wins[player][cell] += reward;
   441→  }
   442→  getRate(player, cell) {
   443→    const v = this.visits[player][cell];
   444→    return v === 0 ? 0.5 : this.wins[player][cell] / v;
   445→  }
   446→  getVisits(player, cell) {
   447→    return this.visits[player][cell];
   448→  }
   449→  decay(factor) {
   450→    for (const p of [1, 2]) {
   451→      for (let i = 0; i < this.visits[p].length; i++) {
   452→        this.visits[p][i] *= factor;
   453→        this.wins[p][i] *= factor;
   454→      }
   455→    }
   456→  }
   457→  totalVisits() {
   458→    let t = 0;
   459→    for (const p of [1, 2]) for (let i = 0; i < this.visits[p].length; i++) t += this.visits[p][i];
   460→    return Math.round(t);
   461→  }
   462→  toJSON() {
   463→    return {
   464→      n: this.n,
   465→      v1: Array.from(this.visits[1]),
   466→      v2: Array.from(this.visits[2]),
   467→      w1: Array.from(this.wins[1]),
   468→      w2: Array.from(this.wins[2]),
   469→    };
   470→  }
   471→  static fromJSON(obj) {
   472→    const g = new GlobalRAVE(obj.n);
   473→    g.visits[1] = Float64Array.from(obj.v1);
   474→    g.visits[2] = Float64Array.from(obj.v2);
   475→    g.wins[1] = Float64Array.from(obj.w1);
   476→    g.wins[2] = Float64Array.from(obj.w2);
   477→    return g;
   478→  }
   479→}

   489→function mulberry32(seed) {
   490→  let s = seed;
   491→  return function () {
   492→    s |= 0;
   493→    s = (s + 0x6d2b79f5) | 0;
   494→    let t = Math.imul(s ^ (s >>> 15), 1 | s);
   495→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
   496→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   497→  };
   498→}

   514→function pickPlayoutMove(board, n, winLen, mode, player, opp, globalRave, rng) {
   515→  const tac = tacticalMove(board, n, winLen, player, opp);
   516→  if (tac) return tac.cell;
   517→  const moves = legalMoves(board);
   518→  if (!moves.length) return null;
   519→  const weights = [];
   520→  let sum = 0;
   521→  for (const cell of moves) {
   522→    const row = Math.floor(cell / n),
   523→      col = cell % n;
   524→    let w;
   525→    if (mode === "classic") {
   526→      const c = classifyCell(board, n, row, col, player, winLen);
   527→      w = 1 + c.score * 0.02;
   528→    } else {
   529→      w = 1 + scoreMoveDelta(board, n, row, col, player) * 3 + scoreMoveDelta(board, n, row, col, opp) * 1.5;
   530→    }
   531→    w *= 0.6 + 0.8 * globalRave.getRate(player, cell);
   532→    weights.push(w);
   533→    sum += w;
   534→  }
   535→  let x = rng() * sum;
   536→  for (let i = 0; i < moves.length; i++) {
   537→    x -= weights[i];
   538→    if (x <= 0) return moves[i];
   539→  }
   540→  return moves[moves.length - 1];
   541→}

   543→function runPlayout(board, n, winLen, mode, startPlayer, globalRave, rng, maxMoves) {
   544→  const work = cloneBoard(board);
   545→  let player = startPlayer;
   546→  const played = [];
   547→  let winner = 0;
   548→  let moves = 0;
   549→  while (moves < maxMoves) {
   550→    const opp = player === 1 ? 2 : 1;
   551→    const cell = pickPlayoutMove(work, n, winLen, mode, player, opp, globalRave, rng);
   552→    if (cell === null) break;
   553→    const row = Math.floor(cell / n),
   554→      col = cell % n;
   555→    work[cell] = player;
   556→    played.push({ player, cell });
   557→    if (mode === "classic" && checkWinAt(work, n, winLen, row, col, player)) {
   558→      winner = player;
   559→      break;
   560→    }
   561→    player = opp;
   562→    moves++;
   563→  }
   564→  let scoreDiff = 0;
   565→  if (mode === "scoring" && legalMoves(work).length === 0) {
   566→    const totals = computeFullScore(work, n);
   567→    scoreDiff = totals[1] - totals[2];
   568→  }
   569→  return { played, winner, scoreDiff };
   570→}

   572→// ---------------------------------------------------------------------------
   573→// Quick Train — fast self-play using the SAME lightweight policy playouts
   574→// use (tactical overrides + threat/score-weighted softmax), with NO tree
   575→// search. This is intentionally cheap so hundreds of games can run in a
   576→// fraction of a second, purely to seed GlobalRAVE before real play.
   577→// Historical bug to avoid (rebuild-spec bug #4): training must apply ZERO
   578→// decay, and playouts must be genuinely randomized, not greedy — both are
   579→// true here because this reuses the exact same runPlayout used inside MCTS.
   580→// ---------------------------------------------------------------------------
   581→export function trainGames(n, winLen, mode, count, globalRave, seedBase = 1) {
   582→  const rng = mulberry32(seedBase);
   583→  const outcomes = { p1: 0, p2: 0, draw: 0 };
   584→  for (let g = 0; g < count; g++) {
   585→    const board = emptyBoard(n);
   586→    const pl = runPlayout(board, n, winLen, mode, 1, globalRave, rng, n * n + 2);
   587→    let reward1; // outcome from player 1's perspective, in [0,1]
   588→    if (mode === "classic") {
   589→      reward1 = pl.winner === 1 ? 1 : pl.winner === 2 ? 0 : 0.5;
   590→      if (pl.winner === 1) outcomes.p1++;
   591→      else if (pl.winner === 2) outcomes.p2++;
   592→      else outcomes.draw++;
   593→    } else {
   594→      reward1 = 1 / (1 + Math.exp(-pl.scoreDiff / Math.max(4, n)));
   595→      if (pl.scoreDiff > 0) outcomes.p1++;
   596→      else if (pl.scoreDiff < 0) outcomes.p2++;
   597→      else outcomes.draw++;
   598→    }
   599→    // THIS is the step that was missing — runPlayout only returns the move
   600→    // list, it doesn't update GlobalRAVE itself (that happens in the
   601→    // caller in runMCTS's simulation step, which trainGames bypasses by
   602→    // calling runPlayout directly). Do it explicitly here.
   603→    for (const mv of pl.played) {
   604→      const r = mv.player === 1 ? reward1 : 1 - reward1;
   605→      globalRave.update(mv.player, mv.cell, r);
   606→    }
   607→  }
   608→  return outcomes;
   609→}

   611→export function runMCTS(opts) {
   612→  const {
   613→    board,
   614→    n,
   615→    winLen,
   616→    mode,
   617→    aiPlayer,
   618→    simulations = 1500,
   619→    timeLimitMs = 1500,
   620→    globalRave,
   621→    seed = 1,
   622→    priorWeight = 0.9,
   623→  } = opts;
   624→  const rng = mulberry32(seed);
   625→  const humanPlayer = aiPlayer === 1 ? 2 : 1;
   626→  const root = new MCTSNode(aiPlayer);
   627→  root.untried = legalMoves(board);
   628→
   629→  const threatPrior = mode === "classic" ? classifyBoard(board, n, aiPlayer, winLen) : null;
   630→  const threatPriorOpp = mode === "classic" ? classifyBoard(board, n, humanPlayer, winLen) : null;
   631→
   632→  const start = Date.now();
   633→  let sims = 0;
   634→  for (; sims < simulations; sims++) {
   635→    if ((sims & 31) === 0 && Date.now() - start > timeLimitMs) break;
   636→
   637→    // ---- Selection ----
   638→    let node = root;
   639→    const state = cloneBoard(board);
   640→    const path = [node];
   641→    let curPlayer = aiPlayer;
   642→    let terminal = false;
   643→    let terminalWinner = 0;
   644→
   645→    while (node.untried && node.untried.length === 0 && node.children.size > 0) {
   646→      let bestCell = null,
   647→        bestScore = -Infinity;
   648→      for (const [cell, child] of node.children) {
   649→        if (state[cell] !== 0) continue; // occupied-cell guard
   650→        const q = child.visits > 0 ? child.wins / child.visits : 0.5;
   651→        const uct = q + Math.SQRT2 * Math.sqrt(Math.log(node.visits + 1) / (child.visits + 1e-6));
   652→        const rv = globalRave.getVisits(node.player, cell);
   653→        const raveRate = globalRave.getRate(node.player, cell);
   654→        const beta = rv > 0 ? raveBeta(child.visits) : 0;
   655→        let score = beta * raveRate + (1 - beta) * uct;
   656→        if (node === root && threatPrior) {
   657→          const tp = curPlayer === aiPlayer ? threatPrior : threatPriorOpp;
   658→          const t = tp.get(cell);
   659→          if (t) {
   660→            const norm = Math.min(1, t.score / CAT_VALUE[CAT.OPEN_FOUR]);
   661→            score += (priorWeight * norm * Math.sqrt(Math.log(node.visits + 2))) / (child.visits + 8);
   662→          }
   663→        }
   664→        if (score > bestScore) {
   665→          bestScore = score;
   666→          bestCell = cell;
   667→        }
   668→      }
   669→      if (bestCell === null) break;
   670→      state[bestCell] = curPlayer;
   671→      const row = Math.floor(bestCell / n),
   672→        col = bestCell % n;
   673→      if (mode === "classic" && checkWinAt(state, n, winLen, row, col, curPlayer)) {
   674→        terminal = true;
   675→        terminalWinner = curPlayer;
   676→      }
   677→      node = node.children.get(bestCell);
   678→      path.push(node);
   679→      curPlayer = curPlayer === 1 ? 2 : 1;
   680→      if (terminal || legalMoves(state).length === 0) break;
   681→    }
   682→
   683→    // ---- Expansion ----
   684→    if (!terminal && legalMoves(state).length > 0) {
   685→      if (node.untried === null) node.untried = legalMoves(state);
   686→      node.untried = node.untried.filter((c) => state[c] === 0);
   687→      if (node.untried.length > 0) {
   688→        const i = Math.floor(rng() * node.untried.length);
   689→        const cell = node.untried[i];
   690→        node.untried.splice(i, 1);
   691→        state[cell] = curPlayer;
   692→        const row = Math.floor(cell / n),
   693→          col = cell % n;
   694→        const child = new MCTSNode(curPlayer === 1 ? 2 : 1);
   695→        node.children.set(cell, child);
   696→        if (mode === "classic" && checkWinAt(state, n, winLen, row, col, curPlayer)) {
   697→          terminal = true;
   698→          terminalWinner = curPlayer;
   699→        }
   700→        node = child;
   701→        path.push(node);
   702→        curPlayer = curPlayer === 1 ? 2 : 1;
   703→      }
   704→    }
   705→
   706→    // ---- Simulation ----
   707→    let reward1; // outcome from PLAYER 1's perspective, in [0,1]
   708→    let playoutMoves = [];
   709→    if (terminal) {
   710→      reward1 = terminalWinner === 1 ? 1 : 0;
   711→    } else if (legalMoves(state).length === 0 && mode === "classic") {
   712→      reward1 = 0.5;
   713→    } else {
   714→      const pl = runPlayout(state, n, winLen, mode, curPlayer, globalRave, rng, n * n + 2);
   715→      playoutMoves = pl.played;
   716→      if (mode === "classic") {
   717→        reward1 = pl.winner === 0 ? 0.5 : pl.winner === 1 ? 1 : 0;
   718→      } else {
   719→        reward1 = 1 / (1 + Math.exp(-pl.scoreDiff / Math.max(4, n)));
   720→      }
   721→    }
   722→    for (const mv of playoutMoves) {
   723→      const r = mv.player === 1 ? reward1 : 1 - reward1;
   724→      globalRave.update(mv.player, mv.cell, r);
   725→    }
   726→
   727→    // ---- Backprop ----
   728→    // Root always tracks value FOR aiPlayer (what "rootWinRate" reports).
   729→    path[0].visits += 1;
   730→    path[0].wins += aiPlayer === 1 ? reward1 : 1 - reward1;
   731→    // Every descendant tracks value for whoever MOVED to reach it, i.e.
   732→    // its parent's `.player` (the mover at the parent, which is exactly
   733→    // who selection compares children on behalf of).
   734→    for (let i = 1; i < path.length; i++) {
   735→      const moverToReachThis = path[i - 1].player;
   736→      path[i].visits += 1;
   737→      path[i].wins += moverToReachThis === 1 ? reward1 : 1 - reward1;
   738→    }
   739→  }
   740→
   741→  let bestCell = null,
   742→    bestVisits = -1;
   743→  const rootChildren = [];
   744→  for (const [cell, child] of root.children) {
   745→    if (board[cell] !== 0) continue;
   746→    rootChildren.push({ cell, visits: child.visits, winRate: child.visits > 0 ? child.wins / child.visits : 0 });
   747→    if (child.visits > bestVisits) {
   748→      bestVisits = child.visits;
   749→      bestCell = cell;
   750→    }
   751→  }
   752→  rootChildren.sort((a, b) => b.visits - a.visits);
   753→
   754→  const override = tacticalMove(board, n, winLen, aiPlayer, humanPlayer);
   755→  if (override && board[override.cell] === 0) bestCell = override.cell;
   756→  if (bestCell === null || board[bestCell] !== 0) {
   757→    const lm = legalMoves(board);
   758→    bestCell = lm[Math.floor(rng() * lm.length)];
   759→  }
   760→
   761→  const rootWinRate = root.visits > 0 ? root.wins / root.visits : 0.5;
   762→  return {
   763→    move: bestCell,
   764→    reason: override ? override.reason : "search",
   765→    simulations: sims,
   766→    elapsedMs: Date.now() - start,
   767→    rootWinRate,
   768→    raveBeta: raveBeta(root.visits),
   769→    topChildren: rootChildren.slice(0, 6),
   770→    totalVisits: root.visits,
   771→  };
   772→}

   841→function countThreats(board, n, player, winLen) {
   842→  const keys = ["SIMPLE_THREE", "BROKEN_THREE", "OPEN_THREE", "SIMPLE_FOUR", "OPEN_FOUR"];
   843→  const counts = Object.fromEntries(keys.map((k) => [k, 0]));
   844→  const cells = classifyBoard(board, n, player, winLen);
   845→  for (const [, c] of cells) {
   846→    for (const k of keys) if (CAT[k] === c.category) counts[k]++;
   847→  }
   848→  return counts;

   851→function buildScoringAnalysis(board, n, scores) {
   852→  const remaining = legalMoves(board).length;
   853→  const filled = n * n - remaining;
   854→  const fillPct = Math.round((filled / (n * n)) * 100);
   855→  const halfFilled = Math.max(1, filled / 2);
   856→  const playerPace = scores[1] / halfFilled;
   857→  const aiPace = scores[2] / halfFilled;
   858→  const halfRemaining = remaining / 2;
   859→  const projPlayer = Math.round(scores[1] + playerPace * halfRemaining);
   860→  const projAi = Math.round(scores[2] + aiPace * halfRemaining);
   861→
   862→  const playerMoves = legalMoves(board)
   863→    .map((cell) => {
   864→      const row = Math.floor(cell / n),
   865→        col = cell % n;
   866→      return { cell, delta: scoreMoveDelta(board, n, row, col, 1) };
   867→    })
   868→    .filter((m) => m.delta > 0)
   869→    .sort((a, b) => b.delta - a.delta)
   870→    .slice(0, 5);
   871→
   872→  const aiThreats = legalMoves(board)
   873→    .map((cell) => {
   874→      const row = Math.floor(cell / n),
   875→        col = cell % n;
   876→      return { cell, delta: scoreMoveDelta(board, n, row, col, 2) };
   877→    })
   878→    .filter((m) => m.delta > 0)
   879→    .sort((a, b) => b.delta - a.delta)
   880→    .slice(0, 5);
   881→
   882→  return { remaining, fillPct, playerPace, aiPace, projPlayer, projAi, playerMoves, aiThreats };
   883→}

   885→function buildAnalysis(board, n, winLen, mode, mctsResult, scores) {
   886→  const ev = evaluatePosition(board, n, winLen);
   887→  const aiWinPctRaw = Math.round(mctsResult.rootWinRate * 100);
   888→  const aiWinPct = Math.max(0, Math.min(100, aiWinPctRaw));
   889→  const playerWinPct = 100 - aiWinPct; // derive, never round both independently (bug #17)
   890→
   891→  const humanCells = classifyBoard(board, n, 1, winLen);
   892→  const humanOptions = [...humanCells.entries()]
   893→    .filter(([, c]) => c.category > CAT.NONE)
   894→    .sort((a, b) => b[1].score - a[1].score)
   895→    .slice(0, 5)
   896→    .map(([cell, c]) => ({ cell, category: c.category, isFork: c.isFork }));
   897→
   898→  const threats = { player: countThreats(board, n, 1, winLen), ai: countThreats(board, n, 2, winLen) };
   899→
   900→  return {
   901→    eval: ev,
   902→    playerWinPct,
   903→    aiWinPct,
   904→    aiReasonCell: mctsResult.move,
   905→    aiReason: mctsResult.reason,
   906→    humanOptions,
   907→    aiConsidered: mctsResult.topChildren,
   908→    threats,
   909→    mcts: {
   910→      simulations: mctsResult.simulations,
   911→      elapsedMs: mctsResult.elapsedMs,
   912→      raveBeta: mctsResult.raveBeta,
   913→      totalVisits: mctsResult.totalVisits,
   914→    },
   915→    scoring: mode === "scoring" ? buildScoringAnalysis(board, n, scores) : null,
   916→  };
   917→}
