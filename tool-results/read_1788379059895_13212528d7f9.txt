     1→   701→        path.push(node);
     2→   702→        curPlayer = curPlayer === 1 ? 2 : 1;
     3→   703→      }
     4→   704→    }
     5→   705→
     6→   706→    // ---- Simulation ----
     7→   707→    let reward1; // outcome from PLAYER 1's perspective, in [0,1]
     8→   708→    let playoutMoves = [];
     9→   709→    if (terminal) {
    10→   710→      reward1 = terminalWinner === 1 ? 1 : 0;
    11→   711→    } else if (legalMoves(state).length === 0 && mode === "classic") {
    12→   712→      reward1 = 0.5;
    13→   713→    } else {
    14→   714→      const pl = runPlayout(state, n, winLen, mode, curPlayer, globalRave, rng, n * n + 2);
    15→   715→      playoutMoves = pl.played;
    16→   716→      if (mode === "classic") {
    17→   717→        reward1 = pl.winner === 0 ? 0.5 : pl.winner === 1 ? 1 : 0;
    18→   718→      } else {
    19→   719→        reward1 = 1 / (1 + Math.exp(-pl.scoreDiff / Math.max(4, n)));
    20→   720→      }
    21→   721→    }
    22→   722→    for (const mv of playoutMoves) {
    23→   723→      const r = mv.player === 1 ? reward1 : 1 - reward1;
    24→   724→      globalRave.update(mv.player, mv.cell, r);
    25→   725→    }
    26→   726→
    27→   727→    // ---- Backprop ----
    28→   728→    // Root always tracks value FOR aiPlayer (what "rootWinRate" reports).
    29→   729→    path[0].visits += 1;
    30→   730→    path[0].wins += aiPlayer === 1 ? reward1 : 1 - reward1;
    31→   731→    // Every descendant tracks value for whoever MOVED to reach it, i.e.
    32→   732→    // its parent's `.player` (the mover at the parent, which is exactly
    33→   733→    // who selection compares children on behalf of).
    34→   734→    for (let i = 1; i < path.length; i++) {
    35→   735→      const moverToReachThis = path[i - 1].player;
    36→   736→      path[i].visits += 1;
    37→   737→      path[i].wins += moverToReachThis === 1 ? reward1 : 1 - reward1;
    38→   738→    }
    39→   739→  }
    40→   740→
    41→   741→  let bestCell = null,
    42→   742→    bestVisits = -1;
    43→   743→  const rootChildren = [];
    44→   744→  for (const [cell, child] of root.children) {
    45→   745→    if (board[cell] !== 0) continue;
    46→   746→    rootChildren.push({ cell, visits: child.visits, winRate: child.visits > 0 ? child.wins / child.visits : 0 });
    47→   747→    if (child.visits > bestVisits) {
    48→   748→      bestVisits = child.visits;
    49→   749→      bestCell = cell;
    50→   750→    }
    51→   751→  }
    52→   752→  rootChildren.sort((a, b) => b.visits - a.visits);
    53→   753→
    54→   754→  const override = tacticalMove(board, n, winLen, aiPlayer, humanPlayer);
    55→   755→  if (override && board[override.cell] === 0) bestCell = override.cell;
    56→   756→  if (bestCell === null || board[bestCell] !== 0) {
    57→   757→    const lm = legalMoves(board);
    58→   758→    bestCell = lm[Math.floor(rng() * lm.length)];
    59→   759→  }
    60→   760→
    61→   761→  const rootWinRate = root.visits > 0 ? root.wins / root.visits : 0.5;
    62→   762→  return {
    63→   763→    move: bestCell,
    64→   764→    reason: override ? override.reason : "search",
    65→   765→    simulations: sims,
    66→   766→    elapsedMs: Date.now() - start,
    67→   767→    rootWinRate,
    68→   768→    raveBeta: raveBeta(root.visits),
    69→   769→    topChildren: rootChildren.slice(0, 6),
    70→   770→    totalVisits: root.visits,
    71→   771→  };
    72→   772→}

   151→   851→function buildScoringAnalysis(board, n, scores) {
   152→   852→  const remaining = legalMoves(board).length;
   153→   853→  const filled = n * n - remaining;
   154→   854→  const fillPct = Math.round((filled / (n * n)) * 100);
   155→   855→  const halfFilled = Math.max(1, filled / 2);
   156→   856→  const playerPace = scores[1] / halfFilled;
   157→   857→  const aiPace = scores[2] / halfFilled;
   158→   858→  const halfRemaining = remaining / 2;
   159→   859→  const projPlayer = Math.round(scores[1] + playerPace * halfRemaining);
   160→   860→  const projAi = Math.round(scores[2] + aiPace * halfRemaining);
   161→   861→
   162→   862→  const playerMoves = legalMoves(board)
   163→   863→    .map((cell) => {
   164→   864→      const row = Math.floor(cell / n),
   165→   865→        col = cell % n;
   166→   866→      return { cell, delta: scoreMoveDelta(board, n, row, col, 1) };
   167→   867→    })
   168→   868→    .filter((m) => m.delta > 0)
   169→   869→    .sort((a, b) => b.delta - a.delta)
   170→   870→    .slice(0, 5);
   171→   871→
   172→   872→  const aiThreats = legalMoves(board)
   173→   873→    .map((cell) => {
   174→   874→      const row = Math.floor(cell / n),
   175→   875→        col = cell % n;
   176→   876→      return { cell, delta: scoreMoveDelta(board, n, row, col, 2) };
   177→   877→    })
   178→   878→    .filter((m) => m.delta > 0)
   179→   879→    .sort((a, b) => b.delta - a.delta)
   180→   880→    .slice(0, 5);
   181→   881→
   182→   882→  return { remaining, fillPct, playerPace, aiPace, projPlayer, projAi, playerMoves, aiThreats };
   183→   883→}

   185→   885→function buildAnalysis(board, n, winLen, mode, mctsResult, scores) {
   186→   886→  const ev = evaluatePosition(board, n, winLen);
   187→   887→  const aiWinPctRaw = Math.round(mctsResult.rootWinRate * 100);
   188→   888→  const aiWinPct = Math.max(0, Math.min(100, aiWinPctRaw));
   189→   889→  const playerWinPct = 100 - aiWinPct; // derive, never round both independently (bug #17)
   190→   890→
   191→   891→  const humanCells = classifyBoard(board, n, 1, winLen);
   192→   892→  const humanOptions = [...humanCells.entries()]
   193→   893→    .filter(([, c]) => c.category > CAT.NONE)
   194→   894→    .sort((a, b) => b[1].score - a[1].score)
   195→   895→    .slice(0, 5)
   196→   896→    .map(([cell, c]) => ({ cell, category: c.category, isFork: c.isFork }));
   197→   897→
   198→   898→  const threats = { player: countThreats(board, n, 1, winLen), ai: countThreats(board, n, 2, winLen) };
   199→   899→
   200→   900→  return {
   201→   901→    eval: ev,
   202→   902→    playerWinPct,
   203→   903→    aiWinPct,
   204→   904→    aiReasonCell: mctsResult.move,
   205→   905→    aiReason: mctsResult.reason,
   206→   906→    humanOptions,
   207→   907→    aiConsidered: mctsResult.topChildren,
   208→   908→    threats,
   209→   909→    mcts: {
   210→   910→      simulations: mctsResult.simulations,
   211→   911→      elapsedMs: mctsResult.elapsedMs,
   212→   912→      raveBeta: mctsResult.raveBeta,
   213→   913→      totalVisits: mctsResult.totalVisits,
   214→   914→    },
   215→   915→    scoring: mode === "scoring" ? buildScoringAnalysis(board, n, scores) : null,
   216→   916→  };
   217→   917→}
