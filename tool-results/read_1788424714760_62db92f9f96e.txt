   261→function MoveReasoning() {

   263→  if (!analysis || moveHistory.length < 2) return null;
   264→  const lastMove = moveHistory[moveHistory.length - 1];
   265→  const colLabel = String.fromCharCode(65 + (lastMove.col >= 8 ? lastMove.col + 1 : lastMove.col));
   266→  const rowLabel = boardSize - lastMove.row;
   267→  const isPlayer = lastMove.player === 1;
   268→  return (
   269→    <div className='space-y-1.5'>
   270→      <div className='flex items-center gap-2'>

   275→            <span className='inline-flex items-center ml-1.5 text-[9px] font-bold text-amber-600 bg-amber-50 px-1.5 py-0.5 rounded-full'>+{lastMove.pointsScored} pts</span>

   319→function StrategicNarrative() {
   320→  const { analysis, moveHistory, boardSize, playerPiece, lastLearning } = useGameStore();
   321→  if (!analysis) return null;
   322→
   323→  const { evalScore, winProb, playerThreats, aiThreats, playerBestMoves, criticalSquares, boardControl, tempo } = analysis;
   324→  const mq = analysis.moveQuality || 'optimal';
   325→  const bc = boardControl || { playerInfluence: 0, aiInfluence: 0, contested: 0, totalCells: 0 };
   326→  const tp = tempo || { playerInitiative: 0, aiInitiative: 0, urgency: 'none' as const };
   327→  const sqs = criticalSquares || [];
   328→
   329→  const totalCells = boardSize * boardSize;
   330→  const played = moveHistory.length;
   331→  const fillPct = Math.round((played / totalCells) * 100);
   332→  const phase = played < totalCells * 0.15 ? 'opening' : played < totalCells * 0.5 ? 'midgame' : 'endgame';
   333→
   334→  const pFours = (playerThreats.OpenFour || 0) + (playerThreats.HalfOpenFour || 0);
   335→  const pThrees = (playerThreats.OpenThree || 0) + (playerThreats.HalfOpenThree || 0);
   336→  const aFours = (aiThreats.OpenFour || 0) + (aiThreats.HalfOpenFour || 0);
   337→  const aThrees = (aiThreats.OpenThree || 0) + (aiThreats.HalfOpenThree || 0);
   338→
   339→  const ctrlTotal = bc.totalCells > 0 ? bc.totalCells : 1;
   340→  const pCtrlPct = Math.round((bc.playerInfluence / ctrlTotal) * 100);
   341→  const aCtrlPct = Math.round((bc.aiInfluence / ctrlTotal) * 100);
   342→
   343→  const sentences: string[] = [];
   344→
   345→  // Phase and fill
   346→  sentences.push(`Move ${played} of ${totalCells} (${fillPct}% filled). ${phase === 'opening' ? 'Early game — center control and pattern foundations are being laid.' : phase === 'midgame' ? 'Midgame — threats are developing and tactical decisions are critical.' : 'Endgame — the board is filling up; converting patterns to points is key.'}`);
   347→
   348→  // Position eval
   349→  if (evalScore > 5) sentences.push('You have a commanding position with strong pattern development. The AI is on the defensive.');
   350→  else if (evalScore > 2) sentences.push('You hold a moderate advantage. Your patterns are developing well, but stay alert to AI counter-play.');
   351→  else if (evalScore > -2) sentences.push('The position is evenly balanced. Both sides have comparable development and opportunities.');
   352→  else if (evalScore > -5) sentences.push('The AI has gained a slight edge. Focus on building connections and disrupting AI patterns.');
   353→  else sentences.push('The AI has a significant advantage. Prioritize blocking threats and look for tactical counter-attacks.');
   354→
   355→  // Threat landscape
   356→  if (pFours > 0 || pThrees > 0) {
   357→    const details: string[] = [];
   358→    if (pFours > 0) details.push(`${pFours} four-in-a-row threat${pFours > 1 ? 's' : ''}`);
   359→    if (pThrees > 0) details.push(`${pThrees} three-in-a-row pattern${pThrees > 1 ? 's' : ''}`);
   360→    sentences.push(`Your offensive position includes ${details.join(' and ')}. ${pFours > 0 ? 'These four-threats demand immediate AI response, giving you initiative.' : 'These patterns can be extended into winning threats with careful play.'}`);
   361→  }
   362→
   363→  if (aFours > 0 || aThrees > 0) {
   364→    const details: string[] = [];
   365→    if (aFours > 0) details.push(`${aFours} four-threat${aFours > 1 ? 's' : ''}`);
   366→    if (aThrees > 0) details.push(`${aThrees} three-pattern${aThrees > 1 ? 's' : ''}`);
   367→    sentences.push(`The AI is building ${details.join(' and ')}. ${aFours > 0 ? 'Four-threats must be blocked immediately or the game is lost.' : 'Monitor these for potential escalation to four-threats next move.'}`);
   368→  }
   369→
   370→  // Board control
   371→  if (pCtrlPct > 60) sentences.push('You dominate board influence, controlling the majority of key positions.');
   372→  else if (aCtrlPct > 60) sentences.push('The AI controls more board territory. Try to contest key zones and build connections in contested areas.');
   373→  else if (bc.contested / ctrlTotal > 0.3) sentences.push('A large portion of the board is contested. Every move here shifts the balance significantly.');
   374→
   375→  // Tempo
   376→  if (tp.urgency === 'critical') sentences.push('CRITICAL: The AI has imminent winning threats. You MUST block now or lose the game.');
   377→  else if (tp.urgency === 'defend') sentences.push('Defensive posture recommended. Block AI patterns before pressing your own attacks.');
   378→  else if (tp.urgency === 'attack') sentences.push('Attack window open! Your threats are strong enough to pressure the AI into defensive moves.');
   379→
   380→  // Recommendation
   381→  if (playerBestMoves.length > 0) {
   382→    const best = playerBestMoves[0];

   388→  if (vitalForAi.length > 0) {
   389→    sentences.push(`Vital defensive squares: ${vitalForAi.map(s => `${coordLabel(s.row, s.col, boardSize)} (${s.reason})`).join(', ')}.`);

   400→            ))}
   401→          </div>
   402→        </div>
   403→      </div>
   404→      {/* Phase indicator badges */}
   405→      <div className='flex items-center gap-2 flex-wrap'>
   406→        <span className={cn('text-[9px] font-bold px-2 py-0.5 rounded-full border',
   407→          phase === 'opening' ? 'bg-emerald-50 text-emerald-700 border-emerald-200' :
   408→          phase === 'midgame' ? 'bg-amber-50 text-amber-700 border-amber-200' :
   409→          'bg-red-50 text-red-700 border-red-200'
   410→        )}>{phase === 'opening' ? '🌱 Opening' : phase === 'midgame' ? '⚔️ Midgame' : '🏁 Endgame'}</span>

   413→        <span className='text-[9px] text-muted-foreground'>{(pThrees + pFours)} vs {(aThrees + aFours)} active threats</span>
   414→      </div>
   415→    </div>
   416→  );
   417→}

   669→function CriticalRow({ sq, boardSize }: { sq: CriticalSquare; boardSize: number }) {
   670→  const severityColor = sq.severity === 'vital' ? 'bg-red-100 text-red-700 border-red-200'

   674→    : sq.forPlayer === 'ai' ? AI_DOT
   675→    : 'bg-gradient-to-br from-amber-300 to-amber-500';
   676→  const forLabel = sq.forPlayer === 'player' ? 'You' : sq.forPlayer === 'ai' ? 'AI' : 'Both';
   677→  return (
   678→    <div className={cn('flex items-center gap-2 px-2 py-1.5 rounded-lg border', severityColor)}>
   679→      <div className={cn('w-2.5 h-2.5 rounded-full shrink-0', forColor)} />

   681→      <span className='flex-1 text-[11px]'>{sq.reason}</span>
   682→      <span className='text-[9px] font-medium opacity-70'>{forLabel}</span>
   683→    </div>
   684→  );
   685→}
